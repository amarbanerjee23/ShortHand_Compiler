cmake_minimum_required(VERSION 3.20)
project(ShortHandMLIRConsumer LANGUAGES C CXX)
find_package(ShortHandMLIR 1 CONFIG REQUIRED)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(SHORTHAND_MLIR_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE ShortHandMLIR::Dialect MLIRParser)
foreach(header IN ITEMS Dialect Types Attributes Ops)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/header_${header}.cpp"
    "#include <ShortHand/IR/ShortHand${header}.h>\n")
  add_library(header_${header} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/header_${header}.cpp")
  target_link_libraries(header_${header} PRIVATE ShortHandMLIR::Dialect)
endforeach()
add_executable(lowering_consumer lowering.cpp)
target_link_libraries(lowering_consumer PRIVATE ShortHandMLIR::Lowering)
foreach(header IN ITEMS SemanticIR Conversion/Lowering)
  string(REPLACE "/" "_" id "${header}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/header_${id}.cpp" "#include <ShortHand/${header}.h>\n")
  add_library(header_${id} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/header_${id}.cpp")
  target_link_libraries(header_${id} PRIVATE ShortHandMLIR::Lowering)
endforeach()
